\section{Mathematical Representation}\label{sec:func}
We describe the model \emph{top-down}. First the whole network is written as one composition of functions (Section~\ref{sec:comp}). Then every function that appears in that composition is defined (Section~\ref{sec:layers}), down to individual neurons and attention heads. The remaining subsections describe the parameters, the outputs, the intermediate states, and the local linear behaviour.
\subsection{The model as a composition of layers}\label{sec:comp}
\begin{definition}[Qwen3.5-0.8B as an $L$-layer decoder]
The model is a function $f_\theta:\mathcal{X}\to\mathcal{Y}$ with parameters $\theta$. It maps a sequence of $T$ token ids $x\in\mathcal{X}=\{1,\dots,V\}^{T}$ to one logit vector per position, $y\in\mathcal{Y}=\mathbb{R}^{T\times V}$ ($V=248{,}320$). It is a composition of an embedding $E_{\mathrm{mm}}$, $L=24$ decoder blocks $\mathcal{B}^{(\ell)}$ and an output head $\Psi$:
\begin{equation}
  f_\theta \;=\; \Psi\circ\mathcal{B}^{(L)}\circ\mathcal{B}^{(L-1)}\circ\cdots\circ\mathcal{B}^{(2)}\circ\mathcal{B}^{(1)}\circ E_{\mathrm{mm}}
\end{equation}
Every block maps $\mathbb{R}^{T\times d}\to\mathbb{R}^{T\times d}$ ($d=1024$), so the blocks can be stacked in any number. Block $\ell$ has its own parameters $\theta_\ell$ and is stored in the checkpoint as \texttt{model.language\_model.layers.}$(\ell-1)$.
\end{definition}
\begin{definition}[Block types]
There are 2 block types (same function form within a type, different weights). A type map $c:\{1,\dots,L\}\to\{\mathrm{L},\mathrm{F}\}$ says which type layer $\ell$ is: $\mathcal{B}^{(\ell)}=\mathcal{B}_{c(\ell)}(\,\cdot\,;\theta_\ell)$. Here $c=((\mathrm{L}^{3}\,\mathrm{F})^{6})$, read left to right from layer 1.
Written out, with the type as a subscript:
\begin{equation}
  f_\theta = \Psi\circ \mathcal{B}^{(24)}_{\mathrm{F}}\circ \cdots\circ \mathcal{B}^{(4)}_{\mathrm{F}}\circ \mathcal{B}^{(3)}_{\mathrm{L}}\circ \mathcal{B}^{(2)}_{\mathrm{L}}\circ \mathcal{B}^{(1)}_{\mathrm{L}}\circ E_{\mathrm{mm}}
\end{equation}
Grouping the repetitions with composition powers ($\mathcal{B}^{\circ n}$ = $n$ consecutive blocks of the same type, each with its own weights) gives the compact form
\begin{equation}\label{eq:compact}
  f_\theta = \Psi\circ\Big(\big(\mathcal{B}_{\mathrm{F}} \circ \mathcal{B}_{\mathrm{L}}^{\circ 3}\big)^{\circ 6}\Big)\circ E_{\mathrm{mm}}
\end{equation}
\end{definition}
\begin{definition}[Vision encoder]
The vision encoder is itself a composition of $12$ blocks acting on a sequence of image patches of width $768$:
\begin{equation}
  \op{Enc_V} = \op{Post}\circ \mathcal{B}_{\mathrm{V}}^{\circ 12}\circ\op{PatchEmbed}
\end{equation}
$\op{PatchEmbed}$ turns the raw input into vectors and $\op{Post}$ is the final norm / pooling / merging (Table~\ref{tab:nonrep}).
\end{definition}

\subsection{The layers, top to bottom}\label{sec:layers}
Every function in the composition above is defined here. Only the attention (or linear-attention) sub-layer mixes information \emph{between} positions $t$. Every other sub-layer (norms, feed-forward networks, projections) acts on each token vector $h_t\in\mathbb{R}^{d}$ independently, e.g.\ $\op{FFN}(h)_t=\op{FFN}(h_t)$. All weights follow the convention $W\in\mathbb{R}^{d_{\mathrm{out}}\times d_{\mathrm{in}}}$, so a projection of a token vector is $W u$ (as in an affine layer $f_i(u)=A^{(i)}u+b^{(i)}$).
\begin{definition}[Token embedding]\label{def:embed}
$E:\{1,\dots,V\}^{T}\to\mathbb{R}^{T\times d}$ looks up one row of the embedding matrix $W_E\in\mathbb{R}^{V\times d}$ per token:
\begin{equation}
  E(x)_t = W_E[x_t,:],\qquad t=1,\dots,T.
\end{equation}
With images, the multimodal embedding replaces the placeholder-token rows of $E(x)$ by projected encoder outputs:
\begin{equation}
  E_{\mathrm{mm}}(x,\cdot) = \op{Merge}\Big(E(x),\ \op{Proj}_{\mathrm{img}}\big(\op{Enc_V}(\mathrm{img})\big)\Big).
\end{equation}
\end{definition}
\begin{definition}[Normalisation]
$\op{RMSNorm}(u)=\gamma\odot u/\sqrt{\tfrac1d\sum_i u_i^2+\epsilon}$, with a learned scale $\gamma\in\mathbb{R}^{d}$, applied to each token vector separately. It fixes the scale of a vector without changing its direction. Every norm in the model has its own $\gamma$, distinguished by a subscript (e.g.\ $\op{RMSNorm}_1$, $\op{RMSNorm}_{\mathrm{final}}$). $\op{LN}$ (LayerNorm, used in some encoder blocks) additionally subtracts the mean and adds a bias.
\end{definition}
\begin{definition}[Block $\mathcal{B}_{\mathrm{L}}$: Linear-attention (Gated DeltaNet) block]
A decoder block of this type maps $h\in\mathbb{R}^{T\times 1024}$ to the same space. It is a composition of 2 residual sub-layers, each of the form $(\mathrm{id}+g)(h)=h+g(h)$:
\begin{equation}
  \mathcal{B}_{\mathrm{L}} = (\mathrm{id}+\op{FFN}\circ \op{RMSNorm}_{2})\circ(\mathrm{id}+\op{Mixer}\circ \op{RMSNorm}_{1})
\end{equation}
Step by step, the sub-layers are:
\begin{align*}
  a &= h + \op{Mixer}(\op{RMSNorm}_{1}(h)) && \text{\footnotesize pre-norm + residual} \\
  \mathcal{B}(h) &= a + \op{FFN}(\op{RMSNorm}_{2}(a)) && \text{\footnotesize pre-norm + residual} \\
  [q,\,k,\,v] &= \op{SiLU}\big(\op{CausalConv1d}(W_{qkv}\,u)\big) && \text{\footnotesize short causal conv (kernel 4) over time} \\
  z &= W_{z}\,u && \text{\footnotesize output gate} \\
  \beta_t &= \sigma(W_{b}\,u_t) && \text{\footnotesize write strength in (0,1)} \\
  g_t &= -e^{A_{\log}}\cdot\op{softplus}(W_{a}\,u_t + b_{dt}) && \text{\footnotesize decay ($\le 0$): how fast memory fades} \\
  q_t,\,k_t &= \op{L2Norm}(q_t),\ \op{L2Norm}(k_t) \\
  S_t &= e^{g_t}\,S_{t-1}\big(I - \beta_t k_t k_t^{\top}\big) + \beta_t\, v_t k_t^{\top} && \text{\footnotesize fixed-size memory per head (delta rule)} \\
  o_t &= S_t\, q_t && \text{\footnotesize read memory with the query} \\
  \op{Mixer}(u) &= W_{\mathrm{out}}\big(\op{RMSNorm}(o)\odot\op{SiLU}(z)\big) \\
  \op{FFN}(u) &= W_{\mathrm{down}}\big(\op{SiLU}(W_{\mathrm{gate}}\,u)\odot W_{\mathrm{up}}\,u\big) && \text{\footnotesize gated MLP, width 3584}
\end{align*}
Parameters of one block ($|\theta_\ell|=21{,}555{,}360$): $K_{\mathrm{conv}}\in\mathbb{R}^{6144\times1\times4}$, $W_{\mathrm{out}}\in\mathbb{R}^{1024\times2048}$, $W_{qkv}\in\mathbb{R}^{6144\times1024}$, $W_z\in\mathbb{R}^{2048\times1024}$, $W_b\in\mathbb{R}^{16\times1024}$, $W_a\in\mathbb{R}^{16\times1024}$, $W_{\mathrm{gate}}\in\mathbb{R}^{3584\times1024}$, $W_{\mathrm{up}}\in\mathbb{R}^{3584\times1024}$, $W_{\mathrm{down}}\in\mathbb{R}^{1024\times3584}$, plus 2,208 parameters in norm scales, biases and other vectors. This type is used 18 times.
\end{definition}
\begin{definition}[Block $\mathcal{B}_{\mathrm{F}}$: Full (global) attention block]
A decoder block of this type maps $h\in\mathbb{R}^{T\times 1024}$ to the same space. It is a composition of 2 residual sub-layers, each of the form $(\mathrm{id}+g)(h)=h+g(h)$:
\begin{equation}
  \mathcal{B}_{\mathrm{F}} = (\mathrm{id}+\op{FFN}\circ \op{RMSNorm}_{2})\circ(\mathrm{id}+\op{Attn}\circ \op{RMSNorm}_{1})
\end{equation}
Step by step, the sub-layers are:
\begin{align*}
  a &= h + \op{Attn}(\op{RMSNorm}_{1}(h)) && \text{\footnotesize pre-norm + residual} \\
  \mathcal{B}(h) &= a + \op{FFN}(\op{RMSNorm}_{2}(a)) && \text{\footnotesize pre-norm + residual} \\
  [Q,\,G] &= W_Q\,u && \text{\footnotesize query and an output gate} \\
  K,\ V &= W_K\,u,\quad W_V\,u && \text{\footnotesize $H_{kv}=2$ key/value heads} \\
  Q,\ K &= \op{RMSNorm}_q(Q),\ \op{RMSNorm}_k(K) && \text{\footnotesize per-head normalisation} \\
  Q,\ K &= \op{RoPE}(Q),\ \op{RoPE}(K) && \text{\footnotesize rotate by position} \\
  S_h &= s\, Q_h K_{g(h)}^{\top} + M && \text{\footnotesize mask $M$: causal} \\
  P_h &= \op{softmax}(S_h) && \text{\footnotesize attention weights} \\
  O_h &= P_h\, V_{g(h)} \\
  \op{Attn}(u) &= W_O\big([O_1;\dots;O_H]\odot\sigma(G)\big) \\
  \op{FFN}(u) &= W_{\mathrm{down}}\big(\op{SiLU}(W_{\mathrm{gate}}\,u)\odot W_{\mathrm{up}}\,u\big) && \text{\footnotesize gated MLP, width 3584}
\end{align*}
Parameters of one block ($|\theta_\ell|=18{,}352{,}640$): $W_Q\in\mathbb{R}^{4096\times1024}$, $W_K\in\mathbb{R}^{512\times1024}$, $W_V\in\mathbb{R}^{512\times1024}$, $W_O\in\mathbb{R}^{1024\times2048}$, $W_{\mathrm{gate}}\in\mathbb{R}^{3584\times1024}$, $W_{\mathrm{up}}\in\mathbb{R}^{3584\times1024}$, $W_{\mathrm{down}}\in\mathbb{R}^{1024\times3584}$, plus 2,560 parameters in norm scales, biases and other vectors. This type is used 6 times.
\end{definition}
\begin{definition}[Block $\mathcal{B}_{\mathrm{V}}$: Vision encoder block]
A vision-encoder block of this type maps $h\in\mathbb{R}^{T\times 768}$ to the same space. It is a composition of 2 residual sub-layers, each of the form $(\mathrm{id}+g)(h)=h+g(h)$:
\begin{equation}
  \mathcal{B}_{\mathrm{V}} = (\mathrm{id}+\op{FFN}\circ \op{LN}_{2})\circ(\mathrm{id}+\op{Attn}\circ \op{LN}_{1})
\end{equation}
Step by step, the sub-layers are:
\begin{align*}
  a &= h + \op{Attn}(\op{LN}_{1}(h)) && \text{\footnotesize pre-norm + residual} \\
  \mathcal{B}(h) &= a + \op{FFN}(\op{LN}_{2}(a)) && \text{\footnotesize pre-norm + residual} \\
  [Q,\,K,\,V] &= W_{qkv}\,u + b_{qkv} && \text{\footnotesize one fused projection} \\
  Q,\ K &= \op{RoPE}_{2D}(Q),\ \op{RoPE}_{2D}(K) && \text{\footnotesize rotate by (row, column) patch position} \\
  S_h &= s\, Q_h K_h^{\top} + M && \text{\footnotesize mask $M$: bidirectional} \\
  P_h &= \op{softmax}(S_h) && \text{\footnotesize attention weights} \\
  O_h &= P_h\, V_h \\
  \op{Attn}(u) &= W_O\,[O_1;\dots;O_H] + b_O \\
  \op{FFN}(u) &= W_2\,\op{GELU}_{\tanh}(W_1 u + b_1) + b_2 && \text{\footnotesize MLP, width 3072}
\end{align*}
Parameters of one block ($|\theta_\ell|=7{,}087{,}872$): $W_{qkv}\in\mathbb{R}^{2304\times768}$, $W_O\in\mathbb{R}^{768\times768}$, $W_1\in\mathbb{R}^{3072\times768}$, $W_2\in\mathbb{R}^{768\times3072}$, plus 9,984 parameters in norm scales, biases and other vectors. This type is used 12 times.
\end{definition}
\begin{definition}[Output head]
$\Psi:\mathbb{R}^{T\times d}\to\mathbb{R}^{T\times V}$ normalises the last hidden state and scores every vocabulary token, re-using the embedding matrix (tied weights):
\begin{equation}
  \Psi(h)_t = W_E^{\top}\,\op{RMSNorm}_{\mathrm{final}}(h)_t
\end{equation}
\end{definition}
\subsection{Neurons and attention heads}
\begin{definition}[Neuron, pre-activation value and state]
The feed-forward sub-layer of block $\ell$ contains $d_{ff}^{(\ell)}=3584$ neurons. Neuron $j$ of layer $\ell$ is the scalar function of a normalised token vector $u\in\mathbb{R}^{d}$
\begin{equation}
  \eta^{(\ell)}_j(u) = \op{SiLU}\big(W^{(\ell)}_{\mathrm{gate},j}\cdot u\big)\;\big(W^{(\ell)}_{\mathrm{up},j}\cdot u\big),
\end{equation}
where $W_{\mathrm{gate},j}$ is the $j$-th row of $W_{\mathrm{gate}}$. The feed-forward output is the sum of the neuron outputs, each written along its own column of $W_{\mathrm{down}}$:
\begin{equation}
  \op{FFN}_\ell(u) = \sum_{j=1}^{d_{ff}^{(\ell)}} \eta^{(\ell)}_j(u)\;W^{(\ell)}_{\mathrm{down}}[:,j].
\end{equation}
Across its $L$ blocks the network has $N=\sum_{\ell=1}^{L} d_{ff}^{(\ell)}=86{,}016$ feed-forward neurons. For input $x$ at position $t$, with $u^{(\ell)}_t(x)$ the normalised input of the feed-forward sub-layer:
\begin{itemize}
\item $\mathcal{V}(\eta;x,t)=W^{(\ell)}_{\mathrm{gate},j}\cdot u^{(\ell)}_t(x)\in\mathbb{R}$ is the \textbf{pre-activation value} of neuron $\eta$ (the argument of the non-linearity).
\item $\mathcal{S}(\eta;x,t)$ is its \textbf{activation state}: \emph{active} if $\mathcal{V}>0$ and \emph{inactive} if $\mathcal{V}\le 0$. The non-linearity is smooth, unlike ReLU, so an inactive neuron's output is strongly suppressed rather than exactly zero.
\end{itemize}
\end{definition}
\begin{definition}[Attention head and attention pattern]
Head $h$ of an attention layer $\ell$ reads from the whole sequence through its attention pattern $P^{(\ell)}_h(x)\in\mathbb{R}^{T\times T}$ (row $t$ = how much position $t$ attends to each earlier position) and writes through its own $d\times d_h$ slice $W^{(\ell)}_{O,h}$ of $W_O$:
\begin{equation}
  \op{Attn}_\ell(u) = \sum_{h=1}^{H} \big(P^{(\ell)}_h V_{g(h)}\odot\sigma(G_h)\big)\,W^{(\ell)\top}_{O,h}.
\end{equation}
$g(h)$ is the key/value head shared by query head $h$ (grouped-query attention). The pattern $P_h$ plays the role that the activation state plays for neurons: once $P_h$ is fixed, the head is linear in its values. The network has 6 attention layers $\times$ $H=8$ heads $=48$ attention heads.
\end{definition}
\subsection{Architecture and parameters}
\begin{definition}[Architecture and parameters]
The \textbf{architecture} is the tuple of sizes and the layer-type sequence:
\begin{equation}
  \mathcal{A} = (V,d,L,c,H,H_{kv},d_h,d_{ff}) = (248{,}320,\ 1024,\ 24,\ ((\mathrm{L}^{3}\,\mathrm{F})^{6}),\ 8,\ 2,\ 256,\ 3584).
\end{equation}
The \textbf{parameters} are $\theta=\{W_E,\ \theta_1,\dots,\theta_L,\ \gamma_{\mathrm{final}},\ \theta_{\op{Enc_V}},\ \theta_{\op{Proj}}\}$ ($W_E$ is shared with the output head). Grouping the blocks by type,
\begin{align*}
  |\theta| &= \sum_{\text{types } c} n_c\,|\theta_c| + |\theta_{\mathrm{rest}}| \\
  &= 18\times21{,}555{,}360 + 6\times18{,}352{,}640 + 12\times7{,}087{,}872 \\
  &\quad+ 269{,}819{,}136 \\
  &= 852{,}985{,}920,
\end{align*}
where $n_c$ is how often type $c$ occurs and $\theta_{\mathrm{rest}}$ holds everything outside the repeated stacks (embedding, final norm, encoder stems and projector, \dots; Table~\ref{tab:nonrep}).
\end{definition}
\subsection{Output representations}
\begin{definition}[Output representations]
For an input sequence $x$, the output can be read at three levels:
\begin{enumerate}
\item \textbf{Logits:} $y=f_\theta(x)\in\mathbb{R}^{T\times V}$, one score per vocabulary token at every position.
\item \textbf{Next-token distribution:} $p(x)_{t,k}=\dfrac{\exp(y_{t,k})}{\sum_{j=1}^{V}\exp(y_{t,j})}=\Pr\big(x_{t+1}=k\mid x_{1},\dots,x_{t}\big)$.
\item \textbf{Predicted next token:} $z(x)=\operatorname{argmax}_{k\in\{1,\dots,V\}} y_{T,k}$ (greedy decoding). Generation appends $z(x)$ (or a sample from $p(x)_{T}$) to $x$ and applies $f_\theta$ again; in practice the keys and values of earlier positions are cached.
\end{enumerate}
The model is \textbf{causal}: $y_t$ depends only on $x_1,\dots,x_t$, because every attention mask hides later positions. Training minimises the cross-entropy $-\sum_{t}\log p(x)_{t,x_{t+1}}$.
\end{definition}
\subsection{Intermediate states and sub-network decomposition}
To study what happens at one layer $\ell\in\{1,\dots,L\}$, split the composition into a prefix, the layer itself and a suffix:
\begin{equation}
  f_\theta = \underbrace{\Psi\circ\mathcal{B}^{(L)}\circ\cdots\circ\mathcal{B}^{(\ell+1)}}_{G_{\ell+1}\ \text{(suffix)}}\circ\underbrace{\mathcal{B}^{(\ell)}}_{\text{layer }\ell}\circ\underbrace{\mathcal{B}^{(\ell-1)}\circ\cdots\circ\mathcal{B}^{(1)}\circ E_{\mathrm{mm}}}_{F_{\ell-1}\ \text{(prefix)}}
\end{equation}
\begin{itemize}
\item \textbf{Prefix} $F_{\ell-1}:\mathcal{X}\to\mathbb{R}^{T\times d}$ maps the tokens to the hidden state entering layer $\ell$ ($F_0=E_{\mathrm{mm}}$).
\item \textbf{Hidden state} (residual stream) after layer $\ell$: $h^{(\ell)}=F_\ell(x)=\mathcal{B}^{(\ell)}(h^{(\ell-1)})\in\mathbb{R}^{T\times d}$, with $h^{(0)}=E_{\mathrm{mm}}(x)$.
\item \textbf{Inside layer $\ell$:} the attention write $a^{(\ell)}=\op{Mixer}_\ell\big(\op{RMSNorm}_{1}(h^{(\ell-1)})\big)$, the mid-block state $\hat h^{(\ell)}=h^{(\ell-1)}+a^{(\ell)}$, the feed-forward input $u^{(\ell)}=\op{RMSNorm}_{2}(\hat h^{(\ell)})$, the pre-activations $\hat x^{(\ell)}=u^{(\ell)}W^{(\ell)\top}_{\mathrm{gate}}\in\mathbb{R}^{T\times d_{ff}}$ and the neuron outputs $\eta^{(\ell)}(u^{(\ell)})\in\mathbb{R}^{T\times d_{ff}}$, and the feed-forward write $m^{(\ell)}=\op{FFN}_\ell(u^{(\ell)})$.
\item \textbf{Suffix} $G_{\ell+1}:\mathbb{R}^{T\times d}\to\mathbb{R}^{T\times V}$ maps $h^{(\ell)}$ to the logits ($G_{L+1}=\Psi$).
\end{itemize}
\paragraph{Residual stream.} Because every sub-layer only \emph{adds} to the hidden state, unrolling the composition gives the final hidden state as the embedding plus everything the layers wrote:
\begin{equation}
  h^{(L)} = h^{(0)} + \sum_{\ell=1}^{L}\big(a^{(\ell)}+m^{(\ell)}\big)
\end{equation}
So $f_\theta(x)=\Psi(h^{(L)})$: each layer reads the current stream and adds a correction to it.
\subsection{Local linear representation}
A ReLU network is piecewise affine: inside a polytope cell it collapses to one matrix. This model is not, because softmax, $\op{SiLU}$ and the normalisations are smooth. Its local behaviour is described by the first-order (Jacobian) linearisation instead. Tokens are discrete, so we perturb the embedded input $h^{(0)}$.
\begin{definition}[Local linear response]
Let $J_\ell=D\mathcal{B}^{(\ell)}(h^{(\ell-1)})$ be the Jacobian of block $\ell$ at the current hidden state (tensors flattened to vectors). The chain rule applied to the residual sub-layers $(\mathrm{id}+g_{\ell,i})$ gives
\begin{equation}
  J_\ell = (I+Dg_{\ell,2})(I+Dg_{\ell,1}),
\end{equation}
where $g_{\ell,1}$ is the attention branch, $g_{\ell,2}$ the feed-forward branch and each $Dg$ is taken at that branch's input. For a small perturbation $\delta$ of $h^{(0)}$, the response of the network decomposed around layer $\ell$ is
\begin{equation}
  f_\theta(h^{(0)}+\delta)-f_\theta(h^{(0)}) \approx G^{(\ell+1)}_x\cdot J_\ell\cdot F^{(\ell-1)}_x\cdot\delta,
\end{equation}
with $F^{(\ell-1)}_x=J_{\ell-1}\cdots J_1$ (prefix) and $G^{(\ell+1)}_x=J_\Psi\,J_L\cdots J_{\ell+1}$ (suffix). Multiplying out the products of $(I+Dg)$ terms writes $J_\Psi J_L\cdots J_1$ as a sum over \emph{paths}: the identity path straight from embedding to head, plus one term for every subset of sub-layers a signal passes through.
\end{definition}

$\op{Enc_V},\ \op{Enc_A},\ \op{Proj}$ & vision / audio encoders and their projectors into the text width \\
$\mathcal{B}^{(\ell)},\ \theta_\ell$ & block $\ell$ and its parameters, $\ell=1,\dots,L$ \\
$\mathcal{B}_c,\ c(\ell)$ & block type $c$; type of layer $\ell$ \\
$\mathcal{B}^{\circ n}$ & $n$ consecutive blocks of one type (each with its own weights) \\
$(\mathrm{id}+g)$ & residual sub-layer $h\mapsto h+g(h)$ \\
$h^{(\ell)}$ & hidden state (residual stream) after layer $\ell$; $h^{(0)}$ = embeddings \\
$a^{(\ell)},\ m^{(\ell)}$ & what the attention / feed-forward sub-layer of layer $\ell$ adds to the stream \\
$\eta^{(\ell)}_j$ & $j$-th feed-forward neuron of layer $\ell$ \\
$\mathcal{V}(\eta;x,t),\ \mathcal{S}(\eta;x,t)$ & pre-activation value and active/inactive state of a neuron \\
$P^{(\ell)}_h$ & attention pattern of head $h$ in layer $\ell$ ($T\times T$) \\
$F_{\ell-1},\ G_{\ell+1}$ & prefix (embedding to layer $\ell-1$) and suffix (layer $\ell+1$ to logits) \\
$J_\ell,\ F^{(\ell-1)}_x,\ G^{(\ell+1)}_x$ & Jacobian of block $\ell$, and of the prefix / suffix at input $x$ \\
$p(x),\ z(x)$ & next-token distribution and greedy next-token prediction \\
\end{longtable}
\section{Forward Pass in Pseudocode}\label{sec:algo}
\begin{algorithm}[H]
\caption{Qwen3.5-0.8B: full forward pass}
\begin{algorithmic}[1]
\Require token ids $x$ of shape $B\times T$; optional images / audio
\State $h \gets E(x)$ \Comment{$h^{(0)}$: one vector per token, $B{\times}T{\times}1024$}
\If{image input present}
\State $e \gets \op{Enc_V}(\text{image})$ \Comment{vision encoder: 12 layers, Algorithm~\ref{alg:vision}}
\State $h[\text{image token slots}] \gets \op{Proj}(e)$ \Comment{project into text width 1024}
\EndIf
\State $\ell \gets 0$
\For{$r \gets 1$ \textbf{to} $6$} \Comment{layers 1--24: pattern of 4 blocks, repeated 6 times}
\For{$j \gets 1$ \textbf{to} $3$}
\State $\ell \gets \ell+1;\ h \gets \mathcal{B}_{\mathrm{L}}(h;\,\theta_\ell)$ \Comment{Linear-attention (Gated DeltaNet) block}
\EndFor
\State $\ell \gets \ell+1;\ h \gets \mathcal{B}_{\mathrm{F}}(h;\,\theta_\ell)$ \Comment{Full (global) attention block}
\EndFor
\State $h \gets \op{RMSNorm}_{\mathrm{final}}(h)$ \Comment{$h=h^{(L)}$; this and the next step are $\Psi$}
\State $y \gets h\,W_E^{\top}$ \Comment{logits $B{\times}T{\times}248,320$, embedding matrix re-used}
\State \Return $y$ \Comment{softmax$(y)$ = next-token probabilities}
\end{algorithmic}
\end{algorithm}
\begin{algorithm}[H]
\caption{Vision encoder}\label{alg:vision}
\begin{algorithmic}[1]
\State $h \gets \op{PatchEmbed}(\text{pixels}) + \text{position embedding}$ \Comment{one vector per image patch}
\For{$\ell \gets 1$ \textbf{to} $12$} \Comment{12 blocks of type $\mathcal{B}_{\mathrm{V}}$}
\State $h \gets \mathcal{B}_{\mathrm{V}}(h;\,\theta_\ell)$ \Comment{Vision encoder block; result is $h^{(\ell)}$}
\EndFor
\State $h \gets \op{Post}(h)$ \Comment{final norm / pooling / merging, see Table~\ref{tab:nonrep}}
\State \Return $h$
\end{algorithmic}
\end{algorithm}
\begin{algorithm}[H]
\caption{Block $\mathcal{B}_{\mathrm{L}}(h)$ --- Linear-attention (Gated DeltaNet) block}
\begin{algorithmic}[1]
\Require $h$ of shape $B\times T\times 1024$
\State $r \gets h$ \Comment{keep the input for the residual (skip) connection}
\State $u \gets \op{RMSNorm}(h)$ \Comment{normalise every token vector}
\Statex \hspace{-1.2em}\textit{token mixing: Gated DeltaNet (linear attention, fixed-size memory)}
\State $q,k,v \gets \op{split}(\op{SiLU}(\op{CausalConv1d}(u W_{qkv}^{\top})))$ \Comment{16 q/k heads of 128, 16 v heads of 128}
\State $z \gets u W_z^{\top};\ \beta \gets \sigma(u W_b^{\top});\ g \gets -e^{A_{\log}}\op{softplus}(u W_a^{\top}+b_{dt})$ \Comment{gate, write strength, decay}
\For{$t \gets 1$ \textbf{to} $T$} \Comment{conceptually sequential; computed in chunks}
\State $S \gets e^{g_t} S\,(I-\beta_t k_t k_t^{\top}) + \beta_t v_t k_t^{\top}$ \Comment{forget a bit, overwrite what $k_t$ points at}
\State $o_t \gets S\, q_t$ \Comment{read}
\EndFor
\State $m \gets \op{RMSNorm}(o)\odot \op{SiLU}(z)\ W_{\mathrm{out}}^{\top}$ \Comment{back to $B{\times}T{\times}1024$}
\State $h \gets r + m$ \Comment{add residual}
\Statex \hspace{-1.2em}\textit{channel mixing: feed-forward network (each token independently)}
\State $r \gets h;\ u \gets \op{RMSNorm}(h)$
\State $m \gets \big(\op{SiLU}(u W_{\mathrm{gate}}^{\top}) \odot u W_{\mathrm{up}}^{\top}\big) W_{\mathrm{down}}^{\top}$ \Comment{expand to 3584, gate, shrink back to 1024}
\State $h \gets r + m$ \Comment{add residual}
\State \Return $h$
\end{algorithmic}
\end{algorithm}
\begin{algorithm}[H]
\caption{Block $\mathcal{B}_{\mathrm{F}}(h)$ --- Full (global) attention block}
\begin{algorithmic}[1]
\Require $h$ of shape $B\times T\times 1024$
\State $r \gets h$ \Comment{keep the input for the residual (skip) connection}
\State $u \gets \op{RMSNorm}(h)$ \Comment{normalise every token vector}
\Statex \hspace{-1.2em}\textit{token mixing: multi-head self-attention}
\State $Q \gets u W_Q^{\top}$ \Comment{$B{\times}T{\times}8{\times}256$ (8 query heads); second half is a gate $G$}
\State $K \gets u W_K^{\top};\ V \gets u W_V^{\top}$ \Comment{$B{\times}T{\times}2{\times}256$ (2 key/value heads, each shared by 4 query heads)}
\State $Q \gets \op{RMSNorm}(Q);\ K \gets \op{RMSNorm}(K)$ \Comment{normalise each head vector (stabilises training)}
\State $Q, K \gets \op{RoPE}(Q, K)$ \Comment{encode position by rotating vector pairs}
\State $P \gets \op{softmax}(s\, Q K^{\top} + M)$ \Comment{mask: causal: a token sees only earlier tokens}
\State $O \gets P V$ \Comment{$B{\times}T{\times}8{\times}256$}
\State $O \gets O \odot \sigma(G)$ \Comment{output gate decides how much of each head to keep}
\State $m \gets O\, W_O^{\top}$ \Comment{concatenate heads, project back to $B{\times}T{\times}1024$}
\State $h \gets r + m$ \Comment{add residual}
\Statex \hspace{-1.2em}\textit{channel mixing: feed-forward network (each token independently)}
\State $r \gets h;\ u \gets \op{RMSNorm}(h)$
\State $m \gets \big(\op{SiLU}(u W_{\mathrm{gate}}^{\top}) \odot u W_{\mathrm{up}}^{\top}\big) W_{\mathrm{down}}^{\top}$ \Comment{expand to 3584, gate, shrink back to 1024}
\State $h \gets r + m$ \Comment{add residual}
\State \Return $h$
\end{algorithmic}
\end{algorithm}
\begin{algorithm}[H]
\caption{Block $\mathcal{B}_{\mathrm{V}}(h)$ --- Vision encoder block}
\begin{algorithmic}[1]
\Require $h$ of shape $B\times T\times 768$
\State $r \gets h$ \Comment{keep the input for the residual (skip) connection}
\State $u \gets \op{LN}(h)$ \Comment{normalise every token vector}
\Statex \hspace{-1.2em}\textit{token mixing: multi-head self-attention}
\State $Q,K,V \gets \op{split}(u W_{qkv}^{\top} + b)$ \Comment{each $B{\times}T{\times}12{\times}64$}
\State $Q, K \gets \op{RoPE}(Q, K)$ \Comment{2-D rotary position (row, column)}
\State $P \gets \op{softmax}(s\, Q K^{\top} + M)$ \Comment{mask: every patch/frame sees every other}
\State $O \gets P V$ \Comment{$B{\times}T{\times}12{\times}64$}
\State $m \gets O\, W_O^{\top} + b_O$ \Comment{concatenate heads, project back to $B{\times}T{\times}768$}
\State $h \gets r + m$ \Comment{add residual}
\Statex \hspace{-1.2em}\textit{channel mixing: feed-forward network (each token independently)}
\State $r \gets h;\ u \gets \op{LN}(h)$
\State $m \gets \op{GELU}_{\tanh}(u W_1^{\top} + b_1) W_2^{\top} + b_2$ \Comment{expand to 3072, shrink back to 768}
\State $h \gets r + m$ \Comment{add residual}
\State \Return $h$
\end{algorithmic}
\end{algorithm}

\multicolumn{2}{l}{\textsc{Language model}} \\
\textbf{Token embedding}\newline{\footnotesize \texttt{embed\_tokens}} & Lookup table with one learned vector per vocabulary token. Turns token ids into the vectors the network computes with (Gemma multiplies them by $\sqrt{d}$). \\
\textbf{Query projection (q\_proj)}\newline{\footnotesize \texttt{self\_attn.\allowbreak{}q\_proj}} & Turns each token vector into query vectors, one per head: ``what am I looking for?''. Here it also outputs an equally sized gate for the attention output (hence the doubled width). \\
\textbf{Query norm (q\_norm)}\newline{\footnotesize \texttt{self\_attn.\allowbreak{}q\_norm}} & RMSNorm applied to every query head vector before RoPE. Keeps attention logits in a sane range (QK-norm). \\
\textbf{Key projection (k\_proj)}\newline{\footnotesize \texttt{self\_attn.\allowbreak{}k\_proj}} & Turns each token into key vectors: ``what do I contain?''. Queries are compared with keys to decide where to attend. With grouped-query attention there are fewer key heads than query heads. \\
\textbf{Key norm (k\_norm)}\newline{\footnotesize \texttt{self\_attn.\allowbreak{}k\_norm}} & RMSNorm applied to every key head vector before RoPE. \\
\textbf{Value projection (v\_proj)}\newline{\footnotesize \texttt{self\_attn.\allowbreak{}v\_proj}} & Turns each token into value vectors: the information actually copied to tokens that attend to it. \\
\textbf{DeltaNet output projection}\newline{\footnotesize \texttt{linear\_attn.\allowbreak{}out\_proj}} & Mixes the heads back into one hidden-size vector. \\
\textbf{Output projection (o\_proj)}\newline{\footnotesize \texttt{self\_attn.\allowbreak{}o\_proj}} & Concatenates all heads' results and mixes them back into one hidden-size vector per token. \\
\textbf{DeltaNet Q/K/V projection}\newline{\footnotesize \texttt{linear\_attn.\allowbreak{}in\_proj\_qkv}} & Produces queries, keys and values for the linear-attention layer in one matrix. \\
\textbf{DeltaNet output-gate projection}\newline{\footnotesize \texttt{linear\_attn.\allowbreak{}in\_proj\_z}} & Produces gate $z$. The normalised output is multiplied by SiLU($z$). \\
\textbf{DeltaNet write-strength projection}\newline{\footnotesize \texttt{linear\_attn.\allowbreak{}in\_proj\_b}} & Produces $\beta=\sigma(\cdot)$ per head: how strongly the current token overwrites the memory. \\
\textbf{DeltaNet decay projection}\newline{\footnotesize \texttt{linear\_attn.\allowbreak{}in\_proj\_a}} & Produces the input-dependent forgetting rate (with A\_log and dt\_bias). \\
\textbf{Short causal convolution}\newline{\footnotesize \texttt{linear\_attn.\allowbreak{}conv1d}} & Depthwise 1-D convolution over the last 4 tokens of Q/K/V. Gives cheap local context before the recurrent memory update. \\
\textbf{A\_log (decay base)}\newline{\footnotesize \texttt{layers.\allowbreak{}linear\_attn}} & Per-head learned log decay rate. $-e^{A_{\log}}$ sets how quickly old memory fades. \\
\textbf{dt\_bias (decay bias)}\newline{\footnotesize \texttt{layers.\allowbreak{}linear\_attn}} & Per-head bias added before the softplus that controls the decay step. \\
\textbf{Gated output RMSNorm}\newline{\footnotesize \texttt{linear\_attn.\allowbreak{}norm}} & Normalises each head's memory read-out and multiplies it by SiLU($z$). \\
\textbf{Gate projection (gate\_proj)}\newline{\footnotesize \texttt{mlp.\allowbreak{}gate\_proj}} & Expands the token vector to the MLP width and passes it through SiLU/GELU. The result acts as a soft on/off switch for each hidden unit. \\
\textbf{Up projection (up\_proj)}\newline{\footnotesize \texttt{mlp.\allowbreak{}up\_proj}} & Expands the token vector to the MLP width (the ``content'' path). It is multiplied element-wise by the gate (SwiGLU/GeGLU). \\
\textbf{Down projection (down\_proj)}\newline{\footnotesize \texttt{mlp.\allowbreak{}down\_proj}} & Shrinks the gated hidden vector back to the model width so it can be added to the residual stream. \\
\textbf{Input norm (input\_layernorm)}\newline{\footnotesize \texttt{layers.\allowbreak{}input\_layernorm}} & RMSNorm applied to the residual stream before attention. Rescales each token vector to unit RMS, then multiplies by a learned per-channel weight. \\
\textbf{post\_attention\_layernorm}\newline{\footnotesize \texttt{layers.\allowbreak{}post\_attention\_layernorm}} & Despite the name, this is the pre-MLP norm: RMSNorm on the residual stream after attention, feeding the MLP. \\
\textbf{Final norm}\newline{\footnotesize \texttt{norm}} & RMSNorm applied once after the last decoder layer, just before the LM head. \\
\midrule \multicolumn{2}{l}{\textsc{Vision encoder}} \\
\textbf{3-D patch embedding (Conv3d)}\newline{\footnotesize \texttt{patch\_embed.\allowbreak{}proj}} & Cuts images/videos into 2-frame $\times$ 16 $\times$ 16-pixel cubes and maps each cube to one vector. \\
\textbf{Position embedding}\newline{\footnotesize \texttt{visual.\allowbreak{}pos\_embed}} & Learned position vectors, interpolated to the image's patch grid. \\
\textbf{Pre-attention LayerNorm}\newline{\footnotesize \texttt{blocks.\allowbreak{}norm1}} & LayerNorm before vision self-attention. \\
\textbf{Pre-MLP LayerNorm}\newline{\footnotesize \texttt{blocks.\allowbreak{}norm2}} & LayerNorm before the vision MLP. \\
\textbf{Fused Q/K/V projection}\newline{\footnotesize \texttt{attn.\allowbreak{}qkv}} & One matrix that produces queries, keys and values at once (split afterwards). \\
\textbf{Attention output projection}\newline{\footnotesize \texttt{attn.\allowbreak{}proj}} & Mixes the concatenated head outputs back to the model width. \\
\textbf{MLP up-projection (fc1)}\newline{\footnotesize \texttt{mlp.\allowbreak{}linear\_fc1}} & Expands each vector to the wider MLP size before the GELU non-linearity. \\
\textbf{MLP down-projection (fc2)}\newline{\footnotesize \texttt{mlp.\allowbreak{}linear\_fc2}} & Shrinks back to the model width. \\
\textbf{Merger LayerNorm}\newline{\footnotesize \texttt{merger.\allowbreak{}norm}} & Normalises patch features before 2$\times$2 neighbouring patches are merged. \\
\textbf{Merger MLP layer 1}\newline{\footnotesize \texttt{merger.\allowbreak{}linear\_fc1}} & Takes 4 neighbouring patches concatenated (2$\times$2 merge, so 4$\times$ fewer image tokens) and mixes them. \\
\textbf{Merger MLP layer 2 (to text width)}\newline{\footnotesize \texttt{merger.\allowbreak{}linear\_fc2}} & Projects merged patches into the language model's hidden size. \\
