\section{Mathematical Representation}\label{sec:func}
We describe the model \emph{top-down}. First the whole network is written as one composition of functions (Section~\ref{sec:comp}). Then every function that appears in that composition is defined (Section~\ref{sec:layers}), down to individual neurons and attention heads. The remaining subsections describe the parameters, the outputs, the intermediate states, and the local linear behaviour.
\subsection{The model as a composition of layers}\label{sec:comp}
\begin{definition}[Llama-3.2-1B as an $L$-layer decoder]
The model is a function $f_\theta:\mathcal{X}\to\mathcal{Y}$ with parameters $\theta$. It maps a sequence of $T$ token ids $x\in\mathcal{X}=\{1,\dots,V\}^{T}$ to one logit vector per position, $y\in\mathcal{Y}=\mathbb{R}^{T\times V}$ ($V=128{,}256$). It is a composition of an embedding $E$, $L=16$ decoder blocks $\mathcal{B}^{(\ell)}$ and an output head $\Psi$:
\begin{equation}
  f_\theta \;=\; \Psi\circ\mathcal{B}^{(L)}\circ\mathcal{B}^{(L-1)}\circ\cdots\circ\mathcal{B}^{(2)}\circ\mathcal{B}^{(1)}\circ E
\end{equation}
Every block maps $\mathbb{R}^{T\times d}\to\mathbb{R}^{T\times d}$ ($d=2048$), so the blocks can be stacked in any number. Block $\ell$ has its own parameters $\theta_\ell$ and is stored in the checkpoint as \texttt{model.layers.}$(\ell-1)$.
\end{definition}
\begin{definition}[Block types]
All $L$ blocks have the same functional form $\mathcal{B}_{\mathrm{D}}$ (Decoder block). They differ only in their weights: $\mathcal{B}^{(\ell)}=\mathcal{B}_{\mathrm{D}}(\,\cdot\,;\theta_\ell)$.
Written out, with the type as a subscript:
\begin{equation}
  f_\theta = \Psi\circ \mathcal{B}^{(16)}_{\mathrm{D}}\circ \cdots\circ \mathcal{B}^{(3)}_{\mathrm{D}}\circ \mathcal{B}^{(2)}_{\mathrm{D}}\circ \mathcal{B}^{(1)}_{\mathrm{D}}\circ E
\end{equation}
Grouping the repetitions with composition powers ($\mathcal{B}^{\circ n}$ = $n$ consecutive blocks of the same type, each with its own weights) gives the compact form
\begin{equation}\label{eq:compact}
  f_\theta = \Psi\circ\Big(\mathcal{B}_{\mathrm{D}}^{\circ 16}\Big)\circ E
\end{equation}
\end{definition}

\subsection{The layers, top to bottom}\label{sec:layers}
Every function in the composition above is defined here. Only the attention (or linear-attention) sub-layer mixes information \emph{between} positions $t$. Every other sub-layer (norms, feed-forward networks, projections) acts on each token vector $h_t\in\mathbb{R}^{d}$ independently, e.g.\ $\op{FFN}(h)_t=\op{FFN}(h_t)$. All weights follow the convention $W\in\mathbb{R}^{d_{\mathrm{out}}\times d_{\mathrm{in}}}$, so a projection of a token vector is $W u$ (as in an affine layer $f_i(u)=A^{(i)}u+b^{(i)}$).
\begin{definition}[Token embedding]\label{def:embed}
$E:\{1,\dots,V\}^{T}\to\mathbb{R}^{T\times d}$ looks up one row of the embedding matrix $W_E\in\mathbb{R}^{V\times d}$ per token:
\begin{equation}
  E(x)_t = W_E[x_t,:],\qquad t=1,\dots,T.
\end{equation}
\end{definition}
\begin{definition}[Normalisation]
$\op{RMSNorm}(u)=\gamma\odot u/\sqrt{\tfrac1d\sum_i u_i^2+\epsilon}$, with a learned scale $\gamma\in\mathbb{R}^{d}$, applied to each token vector separately. It fixes the scale of a vector without changing its direction. Every norm in the model has its own $\gamma$, distinguished by a subscript (e.g.\ $\op{RMSNorm}_1$, $\op{RMSNorm}_{\mathrm{final}}$).
\end{definition}
\begin{definition}[Block $\mathcal{B}_{\mathrm{D}}$: Decoder block]
A decoder block of this type maps $h\in\mathbb{R}^{T\times 2048}$ to the same space. It is a composition of 2 residual sub-layers, each of the form $(\mathrm{id}+g)(h)=h+g(h)$:
\begin{equation}
  \mathcal{B}_{\mathrm{D}} = (\mathrm{id}+\op{FFN}\circ \op{RMSNorm}_{2})\circ(\mathrm{id}+\op{Attn}\circ \op{RMSNorm}_{1})
\end{equation}
Step by step, the sub-layers are:
\begin{align*}
  a &= h + \op{Attn}(\op{RMSNorm}_{1}(h)) && \text{\footnotesize pre-norm + residual} \\
  \mathcal{B}(h) &= a + \op{FFN}(\op{RMSNorm}_{2}(a)) && \text{\footnotesize pre-norm + residual} \\
  Q &= W_Q\,u && \text{\footnotesize $H=32$ heads of size $d_h=64$} \\
  K,\ V &= W_K\,u,\quad W_V\,u && \text{\footnotesize $H_{kv}=8$ key/value heads} \\
  Q,\ K &= \op{RoPE}(Q),\ \op{RoPE}(K) && \text{\footnotesize rotate by position} \\
  S_h &= s\, Q_h K_{g(h)}^{\top} + M && \text{\footnotesize mask $M$: causal} \\
  P_h &= \op{softmax}(S_h) && \text{\footnotesize attention weights} \\
  O_h &= P_h\, V_{g(h)} \\
  \op{Attn}(u) &= W_O\,[O_1;\dots;O_H] \\
  \op{FFN}(u) &= W_{\mathrm{down}}\big(\op{SiLU}(W_{\mathrm{gate}}\,u)\odot W_{\mathrm{up}}\,u\big) && \text{\footnotesize gated MLP, width 8192}
\end{align*}
Parameters of one block ($|\theta_\ell|=60{,}821{,}504$): $W_Q\in\mathbb{R}^{2048\times2048}$, $W_K\in\mathbb{R}^{512\times2048}$, $W_V\in\mathbb{R}^{512\times2048}$, $W_O\in\mathbb{R}^{2048\times2048}$, $W_{\mathrm{gate}}\in\mathbb{R}^{8192\times2048}$, $W_{\mathrm{up}}\in\mathbb{R}^{8192\times2048}$, $W_{\mathrm{down}}\in\mathbb{R}^{2048\times8192}$, plus 4,096 parameters in norm scales, biases and other vectors. This type is used 16 times.
\end{definition}
\begin{definition}[Output head]
$\Psi:\mathbb{R}^{T\times d}\to\mathbb{R}^{T\times V}$ normalises the last hidden state and scores every vocabulary token, re-using the embedding matrix (tied weights):
\begin{equation}
  \Psi(h)_t = W_E^{\top}\,\op{RMSNorm}_{\mathrm{final}}(h)_t
\end{equation}
\end{definition}
\subsection{Neurons and attention heads}
\begin{definition}[Neuron, pre-activation value and state]
The feed-forward sub-layer of block $\ell$ contains $d_{ff}^{(\ell)}=8192$ neurons. Neuron $j$ of layer $\ell$ is the scalar function of a normalised token vector $u\in\mathbb{R}^{d}$
\begin{equation}
  \eta^{(\ell)}_j(u) = \op{SiLU}\big(W^{(\ell)}_{\mathrm{gate},j}\cdot u\big)\;\big(W^{(\ell)}_{\mathrm{up},j}\cdot u\big),
\end{equation}
where $W_{\mathrm{gate},j}$ is the $j$-th row of $W_{\mathrm{gate}}$. The feed-forward output is the sum of the neuron outputs, each written along its own column of $W_{\mathrm{down}}$:
\begin{equation}
  \op{FFN}_\ell(u) = \sum_{j=1}^{d_{ff}^{(\ell)}} \eta^{(\ell)}_j(u)\;W^{(\ell)}_{\mathrm{down}}[:,j].
\end{equation}
Across its $L$ blocks the network has $N=\sum_{\ell=1}^{L} d_{ff}^{(\ell)}=131{,}072$ feed-forward neurons. For input $x$ at position $t$, with $u^{(\ell)}_t(x)$ the normalised input of the feed-forward sub-layer:
\begin{itemize}
\item $\mathcal{V}(\eta;x,t)=W^{(\ell)}_{\mathrm{gate},j}\cdot u^{(\ell)}_t(x)\in\mathbb{R}$ is the \textbf{pre-activation value} of neuron $\eta$ (the argument of the non-linearity).
\item $\mathcal{S}(\eta;x,t)$ is its \textbf{activation state}: \emph{active} if $\mathcal{V}>0$ and \emph{inactive} if $\mathcal{V}\le 0$. The non-linearity is smooth, unlike ReLU, so an inactive neuron's output is strongly suppressed rather than exactly zero.
\end{itemize}
\end{definition}
\begin{definition}[Attention head and attention pattern]
Head $h$ of an attention layer $\ell$ reads from the whole sequence through its attention pattern $P^{(\ell)}_h(x)\in\mathbb{R}^{T\times T}$ (row $t$ = how much position $t$ attends to each earlier position) and writes through its own $d\times d_h$ slice $W^{(\ell)}_{O,h}$ of $W_O$:
\begin{equation}
  \op{Attn}_\ell(u) = \sum_{h=1}^{H} P^{(\ell)}_h\,V_{g(h)}\,W^{(\ell)\top}_{O,h}.
\end{equation}
$g(h)$ is the key/value head shared by query head $h$ (grouped-query attention). The pattern $P_h$ plays the role that the activation state plays for neurons: once $P_h$ is fixed, the head is linear in its values. The network has 16 attention layers $\times$ $H=32$ heads $=512$ attention heads.
\end{definition}
\subsection{Architecture and parameters}
\begin{definition}[Architecture and parameters]
The \textbf{architecture} is the tuple of sizes and the layer-type sequence:
\begin{equation}
  \mathcal{A} = (V,d,L,c,H,H_{kv},d_h,d_{ff}) = (128{,}256,\ 2048,\ 16,\ (\mathrm{D}^{16}),\ 32,\ 8,\ 64,\ 8192).
\end{equation}
The \textbf{parameters} are $\theta=\{W_E,\ \theta_1,\dots,\theta_L,\ \gamma_{\mathrm{final}}\}$ ($W_E$ is shared with the output head). Grouping the blocks by type,
\begin{align*}
  |\theta| &= \sum_{\text{types } c} n_c\,|\theta_c| + |\theta_{\mathrm{rest}}| \\
  &= 16\times60{,}821{,}504 + 262{,}670{,}336 \\
  &= 1{,}235{,}814{,}400,
\end{align*}
where $n_c$ is how often type $c$ occurs and $\theta_{\mathrm{rest}}$ holds everything outside the repeated stacks (embedding, final norm, \dots; Table~\ref{tab:nonrep}).
\end{definition}
\subsection{Output representations}
\begin{definition}[Output representations]
For an input sequence $x$, the output can be read at three levels:
\begin{enumerate}
\item \textbf{Logits:} $y=f_\theta(x)\in\mathbb{R}^{T\times V}$, one score per vocabulary token at every position.
\item \textbf{Next-token distribution:} $p(x)_{t,k}=\dfrac{\exp(y_{t,k})}{\sum_{j=1}^{V}\exp(y_{t,j})}=\Pr\big(x_{t+1}=k\mid x_{1},\dots,x_{t}\big)$.
\item \textbf{Predicted next token:} $z(x)=\operatorname{argmax}_{k\in\{1,\dots,V\}} y_{T,k}$ (greedy decoding). Generation appends $z(x)$ (or a sample from $p(x)_{T}$) to $x$ and applies $f_\theta$ again; in practice the keys and values of earlier positions are cached.
\end{enumerate}
The model is \textbf{causal}: $y_t$ depends only on $x_1,\dots,x_t$, because every attention mask hides later positions. Training minimises the cross-entropy $-\sum_{t}\log p(x)_{t,x_{t+1}}$.
\end{definition}
\subsection{Intermediate states and sub-network decomposition}
To study what happens at one layer $\ell\in\{1,\dots,L\}$, split the composition into a prefix, the layer itself and a suffix:
\begin{equation}
  f_\theta = \underbrace{\Psi\circ\mathcal{B}^{(L)}\circ\cdots\circ\mathcal{B}^{(\ell+1)}}_{G_{\ell+1}\ \text{(suffix)}}\circ\underbrace{\mathcal{B}^{(\ell)}}_{\text{layer }\ell}\circ\underbrace{\mathcal{B}^{(\ell-1)}\circ\cdots\circ\mathcal{B}^{(1)}\circ E}_{F_{\ell-1}\ \text{(prefix)}}
\end{equation}
\begin{itemize}
\item \textbf{Prefix} $F_{\ell-1}:\mathcal{X}\to\mathbb{R}^{T\times d}$ maps the tokens to the hidden state entering layer $\ell$ ($F_0=E$).
\item \textbf{Hidden state} (residual stream) after layer $\ell$: $h^{(\ell)}=F_\ell(x)=\mathcal{B}^{(\ell)}(h^{(\ell-1)})\in\mathbb{R}^{T\times d}$, with $h^{(0)}=E(x)$.
\item \textbf{Inside layer $\ell$:} the attention write $a^{(\ell)}=\op{Attn}_\ell\big(\op{RMSNorm}_{1}(h^{(\ell-1)})\big)$, the mid-block state $\hat h^{(\ell)}=h^{(\ell-1)}+a^{(\ell)}$, the feed-forward input $u^{(\ell)}=\op{RMSNorm}_{2}(\hat h^{(\ell)})$, the pre-activations $\hat x^{(\ell)}=u^{(\ell)}W^{(\ell)\top}_{\mathrm{gate}}\in\mathbb{R}^{T\times d_{ff}}$ and the neuron outputs $\eta^{(\ell)}(u^{(\ell)})\in\mathbb{R}^{T\times d_{ff}}$, and the feed-forward write $m^{(\ell)}=\op{FFN}_\ell(u^{(\ell)})$.
\item \textbf{Suffix} $G_{\ell+1}:\mathbb{R}^{T\times d}\to\mathbb{R}^{T\times V}$ maps $h^{(\ell)}$ to the logits ($G_{L+1}=\Psi$).
\end{itemize}
\paragraph{Residual stream.} Because every sub-layer only \emph{adds} to the hidden state, unrolling the composition gives the final hidden state as the embedding plus everything the layers wrote:
\begin{equation}
  h^{(L)} = h^{(0)} + \sum_{\ell=1}^{L}\big(a^{(\ell)}+m^{(\ell)}\big)
\end{equation}
So $f_\theta(x)=\Psi(h^{(L)})$: each layer reads the current stream and adds a correction to it.
\subsection{Local linear representation}
A ReLU network is piecewise affine: inside a polytope cell it collapses to one matrix. This model is not, because softmax, $\op{SiLU}$ and the normalisations are smooth. Its local behaviour is described by the first-order (Jacobian) linearisation instead. Tokens are discrete, so we perturb the embedded input $h^{(0)}$.
\begin{definition}[Local linear response]
Let $J_\ell=D\mathcal{B}^{(\ell)}(h^{(\ell-1)})$ be the Jacobian of block $\ell$ at the current hidden state (tensors flattened to vectors). The chain rule applied to the residual sub-layers $(\mathrm{id}+g_{\ell,i})$ gives
\begin{equation}
  J_\ell = (I+Dg_{\ell,2})(I+Dg_{\ell,1}),
\end{equation}
where $g_{\ell,1}$ is the attention branch, $g_{\ell,2}$ the feed-forward branch and each $Dg$ is taken at that branch's input. For a small perturbation $\delta$ of $h^{(0)}$, the response of the network decomposed around layer $\ell$ is
\begin{equation}
  f_\theta(h^{(0)}+\delta)-f_\theta(h^{(0)}) \approx G^{(\ell+1)}_x\cdot J_\ell\cdot F^{(\ell-1)}_x\cdot\delta,
\end{equation}
with $F^{(\ell-1)}_x=J_{\ell-1}\cdots J_1$ (prefix) and $G^{(\ell+1)}_x=J_\Psi\,J_L\cdots J_{\ell+1}$ (suffix). Multiplying out the products of $(I+Dg)$ terms writes $J_\Psi J_L\cdots J_1$ as a sum over \emph{paths}: the identity path straight from embedding to head, plus one term for every subset of sub-layers a signal passes through.
\end{definition}

\section{Forward Pass in Pseudocode}\label{sec:algo}
\begin{algorithm}[H]
\caption{Llama-3.2-1B: full forward pass}
\begin{algorithmic}[1]
\Require token ids $x$ of shape $B\times T$
\State $h \gets E(x)$ \Comment{$h^{(0)}$: one vector per token, $B{\times}T{\times}2048$}
\For{$\ell \gets 1$ \textbf{to} $16$} \Comment{16 blocks of type $\mathcal{B}_{\mathrm{D}}$}
\State $h \gets \mathcal{B}_{\mathrm{D}}(h;\,\theta_\ell)$ \Comment{Decoder block; result is $h^{(\ell)}$}
\EndFor
\State $h \gets \op{RMSNorm}_{\mathrm{final}}(h)$ \Comment{$h=h^{(L)}$; this and the next step are $\Psi$}
\State $y \gets h\,W_E^{\top}$ \Comment{logits $B{\times}T{\times}128,256$, embedding matrix re-used}
\State \Return $y$ \Comment{softmax$(y)$ = next-token probabilities}
\end{algorithmic}
\end{algorithm}
\begin{algorithm}[H]
\caption{Block $\mathcal{B}_{\mathrm{D}}(h)$ --- Decoder block}
\begin{algorithmic}[1]
\Require $h$ of shape $B\times T\times 2048$
\State $r \gets h$ \Comment{keep the input for the residual (skip) connection}
\State $u \gets \op{RMSNorm}(h)$ \Comment{normalise every token vector}
\Statex \hspace{-1.2em}\textit{token mixing: multi-head self-attention}
\State $Q \gets u W_Q^{\top}$ \Comment{$B{\times}T{\times}32{\times}64$ (32 query heads)}
\State $K \gets u W_K^{\top};\ V \gets u W_V^{\top}$ \Comment{$B{\times}T{\times}8{\times}64$ (8 key/value heads, each shared by 4 query heads)}
\State $Q, K \gets \op{RoPE}(Q, K)$ \Comment{encode position by rotating vector pairs}
\State $P \gets \op{softmax}(s\, Q K^{\top} + M)$ \Comment{mask: causal: a token sees only earlier tokens}
\State $O \gets P V$ \Comment{$B{\times}T{\times}32{\times}64$}
\State $m \gets O\, W_O^{\top}$ \Comment{concatenate heads, project back to $B{\times}T{\times}2048$}
\State $h \gets r + m$ \Comment{add residual}
\Statex \hspace{-1.2em}\textit{channel mixing: feed-forward network (each token independently)}
\State $r \gets h;\ u \gets \op{RMSNorm}(h)$
\State $m \gets \big(\op{SiLU}(u W_{\mathrm{gate}}^{\top}) \odot u W_{\mathrm{up}}^{\top}\big) W_{\mathrm{down}}^{\top}$ \Comment{expand to 8192, gate, shrink back to 2048}
\State $h \gets r + m$ \Comment{add residual}
\State \Return $h$
\end{algorithmic}
\end{algorithm}

\multicolumn{2}{l}{\textsc{Language model}} \\
\textbf{Token embedding}\newline{\footnotesize \texttt{embed\_tokens}} & Lookup table with one learned vector per vocabulary token. Turns token ids into the vectors the network computes with (Gemma multiplies them by $\sqrt{d}$). \\
\textbf{Query projection (q\_proj)}\newline{\footnotesize \texttt{self\_attn.\allowbreak{}q\_proj}} & Turns each token vector into query vectors, one per head: ``what am I looking for?''. \\
\textbf{Key projection (k\_proj)}\newline{\footnotesize \texttt{self\_attn.\allowbreak{}k\_proj}} & Turns each token into key vectors: ``what do I contain?''. Queries are compared with keys to decide where to attend. With grouped-query attention there are fewer key heads than query heads. \\
\textbf{Value projection (v\_proj)}\newline{\footnotesize \texttt{self\_attn.\allowbreak{}v\_proj}} & Turns each token into value vectors: the information actually copied to tokens that attend to it. \\
\textbf{Output projection (o\_proj)}\newline{\footnotesize \texttt{self\_attn.\allowbreak{}o\_proj}} & Concatenates all heads' results and mixes them back into one hidden-size vector per token. \\
\textbf{Gate projection (gate\_proj)}\newline{\footnotesize \texttt{mlp.\allowbreak{}gate\_proj}} & Expands the token vector to the MLP width and passes it through SiLU/GELU. The result acts as a soft on/off switch for each hidden unit. \\
\textbf{Up projection (up\_proj)}\newline{\footnotesize \texttt{mlp.\allowbreak{}up\_proj}} & Expands the token vector to the MLP width (the ``content'' path). It is multiplied element-wise by the gate (SwiGLU/GeGLU). \\
\textbf{Down projection (down\_proj)}\newline{\footnotesize \texttt{mlp.\allowbreak{}down\_proj}} & Shrinks the gated hidden vector back to the model width so it can be added to the residual stream. \\
\textbf{Input norm (input\_layernorm)}\newline{\footnotesize \texttt{layers.\allowbreak{}input\_layernorm}} & RMSNorm applied to the residual stream before attention. Rescales each token vector to unit RMS, then multiplies by a learned per-channel weight. \\
\textbf{post\_attention\_layernorm}\newline{\footnotesize \texttt{layers.\allowbreak{}post\_attention\_layernorm}} & Despite the name, this is the pre-MLP norm: RMSNorm on the residual stream after attention, feeding the MLP. \\
\textbf{Final norm}\newline{\footnotesize \texttt{norm}} & RMSNorm applied once after the last decoder layer, just before the LM head. \\
